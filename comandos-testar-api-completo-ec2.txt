# COMANDOS PARA TESTAR API COMPLETAMENTE NA EC2
# Execute estes comandos para diagnosticar o problema

# 1. Testar POST (simulação de login)
echo "=== TESTE POST /api/auth/login ==="
curl -v -X POST http://localhost/api/auth/login \
  -H "Content-Type: application/json" \
  -H "Origin: http://18-228-94-6.sslip.io" \
  -d '{"email":"test@test.com","password":"test"}' 2>&1 | head -30
echo ""

# 2. Verificar se há problemas de CORS
echo "=== TESTE OPTIONS (CORS preflight) ==="
curl -v -X OPTIONS http://localhost/api/auth/login \
  -H "Origin: http://18-228-94-6.sslip.io" \
  -H "Access-Control-Request-Method: POST" \
  -H "Access-Control-Request-Headers: content-type" 2>&1 | grep -E "(< HTTP|Access-Control)"
echo ""

# 3. Verificar configuração atual do nginx
echo "=== CONFIGURAÇÃO ATUAL DO NGINX ==="
sudo docker exec geo7-web cat /etc/nginx/conf.d/default.conf
echo ""

# 4. Verificar logs do nginx
echo "=== ÚLTIMOS LOGS DO NGINX ==="
sudo docker logs geo7-web --tail 30 | grep -E "(GET|POST|OPTIONS|error|warn)"
echo ""

# 5. Verificar se o container app está acessível
echo "=== TESTE CONECTIVIDADE APP ==="
sudo docker exec geo7-web wget -O- --timeout=5 http://app:8080/api/healthz 2>&1 | head -5
echo ""
